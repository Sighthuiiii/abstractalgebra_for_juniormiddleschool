\begin{frame}{左边：数着数着，就绕回来了}
  \begin{itemize}
    \item 模 \(12\) 加法：\(8+7=3\)，走 \(7\) 格，绕回来了；
    \item 单位元 \(0\)：加上它等于没加；
    \item 逆元：\(1\) 与 \(11\) 配对、\(2\) 与 \(10\) 配对、\(6\) 与自己配对；
    \item 由 \(3\) 生成的 \(\{0,3,6,9\}\)：只走 \(4\) 步就回到 \(0\)。
  \end{itemize}
  \pause
  \begin{center}
    {\large 数没有变，变的只是「绕圈」的走法。}
  \end{center}
\end{frame}

\begin{frame}{右边：位置换一换，还是那几个样子}
  \begin{itemize}
    \item 正三角形：\(6\) 个对称动作，记作 \(S_3\)；
    \item 正方形：\(8\) 个对称动作，记作 \(D_4\)；
    \item 把复合当作运算：先 \(r\) 后 \(f\)，与先 \(f\) 后 \(r\) 不一样，\(rf\neq fr\)；
    \item 阶与子群：小世界嵌在大世界里，子群的大小整除群的大小。
  \end{itemize}
  \pause
  \begin{center}
    {\large 图形没有变，只是「换个位置摆放」。}
  \end{center}
\end{frame}

\begin{frame}{下方：幂的尾巴，也绕圈}
  \begin{itemize}
    \item \(2^n\) 除以 \(7\)，余数在 \(2,4,1\) 里打转；
    \item 试过许多小素数，归纳出猜想：\(a^{\,p-1}\equiv1\pmod p\)；
    \item 用「余数是重排」把它证出来——费马小定理；
    \item 回头回答第 1 课：\(10^{6}\equiv1\pmod 7\)，所以 \(7\mid999999\)。
  \end{itemize}
\end{frame}

\begin{frame}{三条线汇到一句}
  \begin{theorem}[群]
    一个集合配上一种运算，如果满足下面四条，就叫\textbf{群}：
    \begin{enumerate}
      \item \textbf{封闭}：任取 \(a,b\)，\(a*b\) 还在这里面；
      \item \textbf{结合}：\((a*b)*c=a*(b*c)\)；
      \item \textbf{单位元}：存在 \(e\)，使 \(a*e=e*a=a\)；
      \item \textbf{逆元}：每个 \(a\) 都有 \(b\)，使 \(a*b=b*a=e\)。
    \end{enumerate}
  \end{theorem}
  \pause
  \begin{center}
    {\large 数、动作、置换——只要满足这四条，数学家就把它们归到一起研究。}
  \end{center}
  \pause
  \begin{att}
    这里的 \(*\) 只是个占位的记号：换成 \(+\)、\(\times\)，或者「两个动作接着做」，都行。
  \end{att}
\end{frame}

\section{前面还有什么}

\begin{frame}{加法与乘法，放一起}
  \begin{definition}[环]
    一个集合上同时有加法和乘法，两种运算各自守规矩，又能互相配合，就叫做\textbf{环}。
  \end{definition}
  \pause
  \begin{definition}[域]
    如果环里每个非零元素还能做除法（有乘法逆元），就叫做\textbf{域}。
  \end{definition}
  \pause
  \begin{center}
    整数是环，不是域；有理数是域。
  \end{center}
\end{frame}

\begin{frame}{五次方程的故事}
  \begin{itemize}
    \item 一次、二次方程：有求根公式；
    \item 三次、四次方程：十六世纪也找到了公式；
    \item 五次方程：数学家找了三百多年。
  \end{itemize}
  \pause
  \begin{center}
    {\large 十九世纪，伽罗瓦证明：五次方程\textbf{没有}求根公式。}
  \end{center}
  \pause
  \begin{center}
    {\large 他用的工具，正是这门课讲的——\textbf{群}。\quad 他去世时只有二十岁。}
  \end{center}
\end{frame}
