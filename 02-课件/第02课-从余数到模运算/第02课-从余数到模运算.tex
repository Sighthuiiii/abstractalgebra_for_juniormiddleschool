\section{取余数}

\begin{frame}{今天是星期三，100 天后是星期几？}
  \pause
  \begin{block}{先做一次除法}
    \[
      100 = 7 \times 14 + 2
    \]
  \end{block}
  \pause
  7 天一循环。14 个整星期可以整块扔掉，剩下 2 天。\pause 所以是\textbf{星期五}。
\end{frame}

\begin{frame}{带余除法，你们早就会了}
  \begin{block}{被除数 = 除数 × 商 + 余数}
    \[
      \underbrace{100}_{\text{被除数}} = \underbrace{7}_{\text{除数}} \times \underbrace{14}_{\text{商}} + \underbrace{2}_{\text{余数}}
    \]
  \end{block}
  \pause
  以前我们关心\textbf{商}。\pause 从今天起，我们只关心\textbf{余数}。
\end{frame}

\section{模记号}

\begin{frame}{给「余数」一个记号：mod}
  \[
    100 \equiv 2 \pmod 7
  \]
  读作「100 与 2 对模 7 同余」。
  \pause
  \begin{definition}[同余]
    如果两个整数 \(a,b\) 除以 \(n\) 之后的余数相同，就记作
    \[
      a\equiv b \pmod n,
    \]
    读作「\(a\) 与 \(b\) 对模 \(n\) 同余」。
  \end{definition}
\end{frame}

\begin{frame}{马上用一下}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \(365 \equiv \text{?} \pmod 7\)
        \item \(1000 \equiv \text{?} \pmod 7\)
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \pause
      \begin{itemize}
        \item \(365 = 7\times 52 + 1 \Rightarrow 1\)
        \item \(1000 = 7\times 142 + 6 \Rightarrow 6\)
      \end{itemize}
    \end{column}
  \end{columns}
  \pause
  \vspace{0.6em}
  这两个余数，就是「过 365 天是星期几」和「过 1000 天是星期几」。
\end{frame}

\section{什么是「运算」}

\begin{frame}{先把「运算」这个词说清楚}
  \begin{definition}[运算]
    设 \(S\) 是一个集合。如果有一种办法，把 \(S\) 里任意两个元素 \(a,b\) 合起来，
    得到 \(S\) 里\textbf{唯一确定}的一个元素，记作 \(a*b\)，
    就说 \(*\) 是 \(S\) 上的一个\textbf{运算}。
  \end{definition}
  \pause
  \begin{example}
    有理数加法 \(+\)、有理数乘法 \(\times\)、模 \(12\) 加法——都是运算。
  \end{example}
  \pause
  \begin{center}
    {\large 一句话版：\textbf{任取两个，合起来还是一个，答案唯一}——这就叫运算。}
  \end{center}
\end{frame}

\begin{frame}{写哪个符号，只看方不方便}
  \begin{block}{同一件事，可以写成一堆样子}
    \begin{itemize}
      \item \(3+5\)：数的加法，用 \(+\)
      \item \(3\times5\)、\(3\cdot5\)、\(3*5\)：数的乘法，用 \(\times\)、\(\cdot\) 或 \(*\)
      \item \(3a\)：数字和字母相乘，干脆把乘号省掉不写
      \item \(a*b\)：想强调「这是某套运算的规则」，就用 \(*\)
    \end{itemize}
  \end{block}
  \pause
  \begin{att}
    用哪个符号，只看当下写起来方不方便，\textbf{和规则本身没关系}。
    后面的课里 \(+\)、\(\times\)、\(*\)、\(\oplus\) 都会出现，别被符号吓住。
  \end{att}
\end{frame}

\begin{frame}{运算不是数数才有的东西}
  在有理数上规定一个新规则：\(a*b\) 取 \(a\)、\(b\) 里\textbf{较大的那一个}。
  \begin{example}
    \(3*5=5\)；\quad \((-2)*1=1\)；\quad \(7*7=7\)。
  \end{example}
  \pause
  \begin{block}{它算不算运算？}
    任取两个有理数，合起来结果还是有理数，而且答案唯一——\textbf{算}。
  \end{block}
  \pause
  \begin{center}
    {\large 下节课我们还会看到，\textbf{「动作」也能当运算}。}
  \end{center}
\end{frame}

\section{单位元}

\begin{frame}{哪个数加进去，等于没加？}
  在模 12 的加法里，哪一个数 \(e\) 满足：
  \[
    a + e = a \quad\text{对每个 } a \text{ 都成立}
  \]
  \pause
  \begin{center}
    {\large 是 \textbf{0}。}\quad 因为 \(a+0 = a\)，走 0 格就是不动。
  \end{center}
\end{frame}

\begin{frame}{给它一个正式的名字}
  \begin{definition}[单位元]
    设 \(*\) 是集合 \(S\) 上的一个运算。如果存在 \(e \in S\)，使得
    \[
      a * e = e * a = a
    \]
    对集合里\textbf{每一个}元素 \(a\) 都成立，就称 \(e\) 是这种运算的\textbf{单位元}。
  \end{definition}
  \pause
  \begin{att}
    「单位」是「一个」的意思：它不改变任何东西。
  \end{att}
\end{frame}

\begin{frame}{单位元不止 \(0\) 一个}
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
    运算 & 单位元 & 为什么 \\
    \midrule
    模 \(12\) 加法 & \(0\) & \(a+0=a\)，走 \(0\) 格就是不动 \\
    有理数加法 & \(0\) & \(a+0=a\) \\
    有理数乘法 & \(1\) & \(a\times1=a\) \\
    \bottomrule
  \end{tabular}
  \end{center}
  \pause
  \begin{att}
    加法的单位元是 \(0\)，乘法的单位元是 \(1\)。
    单位元是\textbf{跟着运算走}的，不是永远都叫 \(0\)。
  \end{att}
\end{frame}

\section{逆元}

\begin{frame}{正式的名字：逆元}
  \begin{definition}[逆元]
    设 \(e\) 是运算 \(*\) 的单位元。如果对元素 \(a\) 能找到一个元素 \(b\)，使得
    \[
      a * b = b * a = e
    \]
    就称 \(b\) 是 \(a\) 的\textbf{逆元}，也说 \(a\) 与 \(b\) \textbf{互逆}。
  \end{definition}
  \pause
  \begin{att}
    大白话：逆元就是能把动作\textbf{撤回}的那一步。
  \end{att}
\end{frame}

\begin{frame}{逆元，你们早就在用了}
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
    运算 & 单位元 & \(a\) 的逆元 \\
    \midrule
    有理数加法 & \(0\) & 相反数 \(-a\) \\
    有理数乘法 & \(1\) & 倒数 \(\dfrac{1}{a}\ (a\neq 0)\) \\
    模 \(12\) 加法 & \(0\) & ？ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \pause
  「加上 \(3\)」的逆动作是「加上 \(-3\)」；「乘上 \(3\)」的逆动作是「乘上 \(\frac{1}{3}\)」——
  \pause
  \begin{center}
    {\large 模 \(12\) 加法里的逆元长什么样？\textbf{下节课先把它找齐。}}
  \end{center}
\end{frame}
