\section{八个动作}

\begin{frame}{一张正方形纸片}
  四个顶点写上 \(1,2,3,4\)：\quad \(1\) 左上，\(2\) 右上，\(3\) 右下，\(4\) 左下。
  \imgph[3.2cm]{正方形纸片：正面写 \(1,2,3,4\)（顺时针），背面涂成另一种颜色}
\end{frame}

\begin{frame}{一共八个}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{转动（\(4\) 个）}
        \begin{itemize}
          \item 不动
          \item 顺时针转 \(90^\circ\)
          \item 顺时针转 \(180^\circ\)
          \item 顺时针转 \(270^\circ\)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{翻折（\(4\) 个）}
        \begin{itemize}
          \item 沿对角线 \(1\text{—}3\) 翻
          \item 沿对角线 \(2\text{—}4\) 翻
          \item 沿竖直中线翻
          \item 沿水平中线翻
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \pause
  \begin{center}
    \(4+4=8\)。\quad 没有第九个。
  \end{center}
\end{frame}

\begin{frame}{跟三角形比一比}
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
     & 正三角形 & 正方形 \\
    \midrule
    转动 & \(3\) 个 & \(4\) 个 \\
    翻折 & \(3\) 个 & \(4\) 个 \\
    一共 & \(6\) 个 & \(8\) 个 \\
    \midrule
    转 \(90^\circ\) 要连做几次才回来 & —— & \(\mathbf{4}\) 次 \\
    转 \(120^\circ\) 要连做几次才回来 & \(\mathbf{3}\) 次 & —— \\
    \bottomrule
  \end{tabular}
  \end{center}
  \pause
  \begin{center}
    {\large 转 \(90^\circ\) 的阶是 \(4\)；三角形的「转 \(120^\circ\)」阶是 \(3\)。}
  \end{center}
\end{frame}

\begin{frame}{八个动作，八张两行式（转动）}
  取名字：\(r\) 是「顺时针转 \(90^\circ\)」，\(f\) 是「沿对角线 \(1\text{—}3\) 翻折」。
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
    名字 & 动作 & 两行式 \\
    \midrule
    \(e\) & 不动 & \(\begin{pmatrix}1&2&3&4\\1&2&3&4\end{pmatrix}\) \\[6pt]
    \(r\) & 转 \(90^\circ\) & \(\begin{pmatrix}1&2&3&4\\2&3&4&1\end{pmatrix}\) \\[6pt]
    \(r^2\) & 转 \(180^\circ\) & \(\begin{pmatrix}1&2&3&4\\3&4&1&2\end{pmatrix}\) \\[6pt]
    \(r^3\) & 转 \(270^\circ\) & \(\begin{pmatrix}1&2&3&4\\4&1&2&3\end{pmatrix}\) \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{八个动作，八张两行式（翻折）}
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
    名字 & 动作 & 两行式 \\
    \midrule
    \(f\) & 沿对角线 \(1\text{—}3\) 翻 & \(\begin{pmatrix}1&2&3&4\\1&4&3&2\end{pmatrix}\) \\[6pt]
    \(rf\) & 沿水平中线翻 & \(\begin{pmatrix}1&2&3&4\\4&3&2&1\end{pmatrix}\) \\[6pt]
    \(r^2f\) & 沿对角线 \(2\text{—}4\) 翻 & \(\begin{pmatrix}1&2&3&4\\3&2&1&4\end{pmatrix}\) \\[6pt]
    \(r^3f\) & 沿竖直中线翻 & \(\begin{pmatrix}1&2&3&4\\2&1&4&3\end{pmatrix}\) \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{用手指回答}
  \begin{itemize}
    \item 正方形一共几个对称动作？
    \item \(r\)（转 \(90^\circ\)）的阶是多少？
    \item \(f\)（沿对角线翻）的阶是多少？
    \item \(r^2\)（转 \(180^\circ\)）的阶是多少？
  \end{itemize}
  \pause
  \begin{block}{答案}
    \(8\)；\quad \(4\)；\quad \(2\)；\quad \(2\)。
  \end{block}
\end{frame}

\section{乘法表}

\begin{frame}{乘法表：还是只填两行}
  横竖都按 \(e,r,r^2,r^3,f,rf,r^2f,r^3f\) 排。格子里的动作是「\textbf{先横排的，再做竖排的}」。
  \imgph[3.0cm]{\(8\times 8\) 乘法表：只填好 \(r\) 行与 \(f\) 行，其余留空（课后贴教室墙）}
  \[
  \begin{array}{@{}l|cccccccc@{}}
    \toprule
     & e & r & r^2 & r^3 & f & rf & r^2f & r^3f \\
    \midrule
    \text{先 }r & r & r^2 & r^3 & e & rf & r^2f & r^3f & f \\
    \text{先 }f & f & r^3f & r^2f & rf & e & r^3 & r^2 & r \\
    \bottomrule
  \end{array}
  \]
\end{frame}

\begin{frame}{每一行还是重排}
  \begin{block}{看这两行}
    \(r\) 行：\(r,\ r^2,\ r^3,\ e,\ rf,\ r^2f,\ r^3f,\ f\)\\
    \(f\) 行：\(f,\ r^3f,\ r^2f,\ rf,\ e,\ r^3,\ r^2,\ r\)
  \end{block}
  \pause
  \begin{center}
    {\large 两行都是同样 \(8\) 个动作，一个不多，一个不少。}
  \end{center}
\end{frame}

\begin{frame}{两个关系，跟三角形那两条一样}
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
    \toprule
     & 三角形 \(S_3\) & 正方形 \(D_4\) \\
    \midrule
    基本转动 & \(r\)（转 \(120^\circ\)） & \(r\)（转 \(90^\circ\)） \\
    连做几次回来 & \(3\) 次 & \(4\) 次 \\
    翻折的关系式 & \(fr=r^2f\) & \(fr=r^3f\) \\
    \bottomrule
  \end{tabular}
  \end{center}
  \pause
  \begin{center}
    {\large 两边都是「先翻后转 \(=\) 先\textbf{倒着}转再翻」。}
  \end{center}
\end{frame}

\begin{frame}{这节课要带走的三句话}
  \begin{itemize}
    \item 正方形的对称动作一共 \(8\) 个：\(4\) 个转动加 \(4\) 个翻折。
    \item 每个动作都能写成两行式，做法和三角形完全一样。
    \item 乘法表每一行、每一列还是那 \(8\) 个动作的一次重排。
  \end{itemize}

  \end{center}
\end{frame}
