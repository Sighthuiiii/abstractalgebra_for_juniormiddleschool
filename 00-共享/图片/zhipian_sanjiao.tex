    \tricard{\(A\)}{\(B\)}{\(C\)}

      \node[text=tuback] at (0.49,-0.29)  {\tufont \(B\)};%
      \node[text=tuback] at (-0.49,-0.29) {\tufont \(C\)};%
    \end{scope}
  \end{scope}

  % ---- 中间：翻折箭头 ----
  \draw[tuarrb] (-1.90,0.95) .. controls (-0.90,2.00) and (0.90,2.00) .. (1.90,0.95);
  \node at (0,2.30) {\tufonts 翻过来};

  % ---- 标注 ----
