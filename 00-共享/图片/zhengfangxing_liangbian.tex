% 图 10：正方形两次演示对照 —— 第 09 课 f9
% 左：先翻 f（沿对角线 1—3），再转 90°（r），即 fr
% 右：先转 270°（r^3），再翻 f，即 r^3f
% 两条路都落在同一个位置（fr = r^3f），所以两个终点的卡完全一样
% 卡内写的是此刻各顶点上挂的是原来的哪个编号
\documentclass[tikz,border=2pt]{standalone}

\input{common.tex}

% 正方形卡：#1#2#3#4 = 左上/右上/右下/左下 位置上的标签；#5 = 标签颜色
\newcommand{\scard}[5]{%
  \begin{scope}[scale=\sqS]
    \draw[tuline] (-1,1) -- (1,1) -- (1,-1) -- (-1,-1) -- cycle;
    \node[text=#5] at (-0.52,0.52)  {\tufont #1};%
    \node[text=#5] at (0.52,0.52)   {\tufont #2};%
    \node[text=#5] at (0.52,-0.52)  {\tufont #3};%
    \node[text=#5] at (-0.52,-0.52) {\tufont #4};%
  \end{scope}%
}

\begin{document}
\begin{tikzpicture}
% ================= 左：先翻 f，再转 r =================
\node at (0,2.15) {\tufonts 先翻 \(f\)，再转 \(90^\circ\)（\(r\)）};
\begin{scope}[shift={(-3.30,0)}] \scard{\(1\)}{\(2\)}{\(3\)}{\(4\)}{black}  \end{scope}
\begin{scope}[shift={(0,0)}]     \scard{\(1\)}{\(4\)}{\(3\)}{\(2\)}{tuback} \end{scope}
\begin{scope}[shift={(3.30,0)}]  \scard{\(3\)}{\(2\)}{\(1\)}{\(4\)}{tuhigh} \end{scope}
\draw[tuarrb] (-1.82,0) -- (-1.48,0);
\draw[tuarrb] (1.48,0)  -- (1.82,0);
\node at (-3.30,-2.00) {\tufonts 起点};
\node at (0,-2.00)     {\tufonts 做完 \(f\)};
\node at (3.30,-2.00)  {\tufonts 终点};

% ================= 右：先转 r^3，再翻 f =================
\node at (9.60,2.15) {\tufonts 先转 \(270^\circ\)（\(r^3\)），再翻 \(f\)};
\begin{scope}[shift={(6.30,0)}]   \scard{\(1\)}{\(2\)}{\(3\)}{\(4\)}{black}  \end{scope}
\begin{scope}[shift={(9.60,0)}]   \scard{\(4\)}{\(1\)}{\(2\)}{\(3\)}{tuback} \end{scope}
\begin{scope}[shift={(12.90,0)}]  \scard{\(3\)}{\(2\)}{\(1\)}{\(4\)}{tuhigh} \end{scope}
\draw[tuarrb] (7.88,0)  -- (8.08,0);
\draw[tuarrb] (11.18,0) -- (11.38,0);
\node at (6.30,-2.00)  {\tufonts 起点};
\node at (9.60,-2.00)  {\tufonts 做完 \(r^3\)};
\node at (12.90,-2.00) {\tufonts 终点};
\end{tikzpicture}
