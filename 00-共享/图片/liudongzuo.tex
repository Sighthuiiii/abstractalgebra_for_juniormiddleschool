% 图 5：六个动作总览 —— 第 05 课 f6（本套材料最重要的一张）
% 两行三列：第一行 不动 / 转 120° / 转 240°；第二行 三个翻折（背面颜色）
% 每个三角形上的字母 = 原来的 A/B/C 各走到了哪个顶点
% 第一格（不动）不写字母，它就是“标准摆法”本身
\documentclass[tikz,border=2pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\input{common.tex}

% 正面三角形，三个顶点上写 #1#2#3（上、右下、左下）
\newcommand{\trilab}[3]{%
  \begin{scope}[scale=\triS]
    \draw[tuline] (0,1) -- (-0.86603,-0.5) -- (0.86603,-0.5) -- cycle;
    \node at (-0.405,0.315) {\tufont #1};%
    \node at (0.49,-0.29)   {\tufont #2};%
    \node at (-0.49,-0.29)  {\tufont #3};%
  \end{scope}%
}
% 背面三角形（填充背面色），三个顶点上写 #1#2#3
\newcommand{\trilabb}[3]{%
  \begin{scope}[scale=\triS]
    \fill[tubfill] (0,1) -- (-0.86603,-0.5) -- (0.86603,-0.5) -- cycle;
    \draw[tuback] (0,1) -- (-0.86603,-0.5) -- (0.86603,-0.5) -- cycle;
    \node[text=tuback] at (-0.405,0.315) {\tufont #1};%
    \node[text=tuback] at (0.49,-0.29)   {\tufont #2};%
    \node[text=tuback] at (-0.49,-0.29)  {\tufont #3};%
  \end{scope}%
}

\begin{document}
\begin{tikzpicture}
  % ---------- 第一行：不动、转 120°、转 240° ----------
  \node at (0,1.60)      {\tufonts 不动};
  \node at (3.90,1.60)   {\tufonts 转 \(120^\circ\)};
  \node at (7.80,1.60)   {\tufonts 转 \(240^\circ\)};
  \begin{scope}[shift={(0,0)}]     \trilab{\(A\)}{\(B\)}{\(C\)} \end{scope}
  \begin{scope}[shift={(3.90,0)}]  \trilab{\(B\)}{\(C\)}{\(A\)} \end{scope}
  \begin{scope}[shift={(7.80,0)}]  \trilab{\(C\)}{\(A\)}{\(B\)} \end{scope}

  % ---------- 第二行：过 A / B / C 的轴翻（背面颜色）----------
  \node at (0,-2.35)      {\tufonts 过 \(A\) 的轴翻};
  \node at (3.90,-2.35)   {\tufonts 过 \(B\) 的轴翻};
  \node at (7.80,-2.35)   {\tufonts 过 \(C\) 的轴翻};
  \begin{scope}[shift={(0,-3.95)}]     \trilabb{\(A\)}{\(C\)}{\(B\)} \end{scope}
  \begin{scope}[shift={(3.90,-3.95)}]  \trilabb{\(C\)}{\(B\)}{\(A\)} \end{scope}
  \begin{scope}[shift={(7.80,-3.95)}]  \trilabb{\(B\)}{\(A\)}{\(C\)} \end{scope}
\end{tikzpicture}
\end{document}
