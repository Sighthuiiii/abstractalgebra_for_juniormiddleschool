% 普通三角形卡：#1#2#3 = A/B/C 位置上的标签；#4 = 标签颜色
\newcommand{\tcard}[4]{%
  \begin{scope}[scale=\triS]
    \draw[tuline] (0,1) -- (-0.86603,-0.5) -- (0.86603,-0.5) -- cycle;
    \node[text=#4] at (0,0.55)      {\tufont #1};%
    \node[text=#4] at (0.49,-0.29)  {\tufont #2};%
    \node[text=#4] at (-0.49,-0.29) {\tufont #3};%
  \end{scope}%
}
% 带背面颜色的三角形卡
\newcommand{\tcardb}[3]{%
  \begin{scope}[scale=\triS]
    \fill[tubfill] (0,1) -- (-0.86603,-0.5) -- (0.86603,-0.5) -- cycle;
    \draw[tuback] (0,1) -- (-0.86603,-0.5) -- (0.86603,-0.5) -- cycle;
    \node[text=tuback] at (0,0.55)      {\tufont #1};%
    \node[text=tuback] at (0.49,-0.29)  {\tufont #2};%
    \node[text=tuback] at (-0.49,-0.29) {\tufont #3};%
  \end{scope}%
}

\begin{tikzpicture}
% ================= 左：先 f 后 r =================
\node at (0,1.90) {\tufonts 先做 \(f\)，再做 \(r\)};\begin{scope}[shift={(-3.00,0)}] \tcard{\(A\)}{\(B\)}{\(C\)}{black}  \end{scope}
\begin{scope}[shift={(0,0)}]     \tcardb{\(A\)}{\(C\)}{\(B\)}        \end{scope}
